import React, { useState, useRef, useEffect } from 'react';
import { Card, CardHeader, CardTitle, CardContent, CardFooter } from '@/components/ui/card';
import { Label } from '@/components/ui/label';
import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Alert, AlertDescription } from '@/components/ui/alert';
import { CheckCircle2 } from 'lucide-react';

export default function App() {
  const [selectedValue, setSelectedValue] = useState('A');
  const [customText, setCustomText] = useState('');
  const [isSubmitted, setIsSubmitted] = useState(false);
  const inputRef = useRef(null);

  const options = [
    { id: 'A', label: '选项 A: 确定性方案一' },
    { id: 'B', label: '选项 B: 确定性方案二' },
    { id: 'C', label: '选项 C: 确定性方案三' },
  ];

  // 自动聚焦处理
  useEffect(() => {
    if (selectedValue === 'D' && inputRef.current) {
      inputRef.current.focus();
    }
  }, [selectedValue]);

  const handleSubmit = (e) => {
    e.preventDefault();
    const finalData = {
      choice: selectedValue,
      content: selectedValue === 'D' ? customText : options.find(o => o.id === selectedValue)?.label
    };
    
    if (selectedValue === 'D' && !customText.trim()) {
      return; 
    }

    console.log('提交的数据:', finalData);
    setIsSubmitted(true);
    setTimeout(() => setIsSubmitted(false), 3000);
  };

  return (
    <div className="flex items-center justify-center min-h-screen bg-slate-50 p-4 font-sans">
      <Card className="w-full max-w-md shadow-xl border-slate-200">
        <CardHeader className="pb-4">
          <CardTitle className="text-xl font-bold text-slate-900 text-center">
            方案意向收集
          </CardTitle>
        </CardHeader>
        
        <form onSubmit={handleSubmit}>
          <CardContent className="space-y-4">
            <RadioGroup 
              value={selectedValue} 
              onValueChange={setSelectedValue}
              className="gap-0 -space-y-px rounded-lg border border-slate-200 overflow-hidden"
            >
              {options.map((option) => (
                <div 
                  key={option.id}
                  className={`flex items-center space-x-3 p-4 border-b last:border-b-0 transition-colors cursor-pointer ${
                    selectedValue === option.id ? 'bg-slate-50' : 'hover:bg-slate-50/50'
                  }`}
                  onClick={() => setSelectedValue(option.id)}
                >
                  <RadioGroupItem value={option.id} id={option.id} />
                  <Label 
                    htmlFor={option.id} 
                    className="flex-grow cursor-pointer text-sm font-medium text-slate-700"
                  >
                    {option.label}
                  </Label>
                </div>
              ))}

              {/* 选项 D: 无标题集成输入框 */}
              <div 
                className={`flex items-center space-x-3 p-4 transition-colors cursor-pointer ${
                  selectedValue === 'D' ? 'bg-slate-50' : 'hover:bg-slate-50/50'
                }`}
                onClick={() => setSelectedValue('D')}
              >
                <RadioGroupItem value="D" id="D" />
                <div className="flex-grow">
                  {selectedValue === 'D' ? (
                    <Input
                      ref={inputRef}
                      placeholder="请输入自定义内容..."
                      value={customText}
                      onChange={(e) => setCustomText(e.target.value)}
                      className="h-8 text-sm bg-white border-slate-300 focus-visible:ring-slate-900"
                      onClick={(e) => e.stopPropagation()} 
                    />
                  ) : (
                    <span className="text-sm font-medium text-slate-400">
                      选项 D: 点击输入自定义内容
                    </span>
                  )}
                </div>
              </div>
            </RadioGroup>

            {isSubmitted && (
              <Alert className="bg-slate-900 text-white border-none py-3">
                <CheckCircle2 className="h-4 w-4 stroke-white" />
                <AlertDescription className="text-xs">
                  提交成功。
                </AlertDescription>
              </Alert>
            )}
          </CardContent>

          <CardFooter>
            <Button 
              type="submit" 
              className="w-full bg-slate-900 hover:bg-slate-800 text-white h-11 transition-all"
              disabled={selectedValue === 'D' && !customText.trim()}
            >
              确认提交
            </Button>
          </CardFooter>
        </form>
      </Card>
    </div>
  );
}
