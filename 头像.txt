import React, { useState, useEffect, useRef } from 'react';
import { 
  Send, 
  Plus, 
  ChevronDown, 
  MessageSquare,
  Trash2
} from 'lucide-react';

const APP_STORAGE_KEY = 'copilot_sessions_v2';

const App = () => {
  const [sessions, setSessions] = useState([]);
  const [activeSessionId, setActiveSessionId] = useState(null);
  const [showHistoryDropdown, setShowHistoryDropdown] = useState(false);
  const [showPersonaDropdown, setShowPersonaDropdown] = useState(false);
  const [activePersona, setActivePersona] = useState('人格导师');
  const [input, setInput] = useState('');
  const [isTyping, setIsTyping] = useState(false);
  
  const messagesEndRef = useRef(null);
  const textareaRef = useRef(null);

  const personas = ['人格面具 (Persona)', '暗影 (Shadow)', '阿尼玛 (Anima)', '阿尼姆斯 (Animus)', '自性 (Self)'];

  useEffect(() => {
    const saved = localStorage.getItem(APP_STORAGE_KEY);
    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        setSessions(parsed);
        if (parsed.length > 0) setActiveSessionId(parsed[0].id);
        else createNewSession();
      } catch (e) { createNewSession(); }
    } else { createNewSession(); }
  }, []);

  useEffect(() => {
    if (sessions.length > 0) {
      localStorage.setItem(APP_STORAGE_KEY, JSON.stringify(sessions));
    }
  }, [sessions]);

  useEffect(() => {
    if (textareaRef.current) {
      textareaRef.current.style.height = 'auto';
      const scrollHeight = textareaRef.current.scrollHeight;
      const clampedHeight = Math.min(Math.max(scrollHeight, 44), 300);
      textareaRef.current.style.height = `${clampedHeight}px`;
      textareaRef.current.style.overflowY = scrollHeight >= 300 ? 'auto' : 'hidden';
    }
  }, [input]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [sessions, activeSessionId, isTyping]);

  const createNewSession = () => {
    const newSession = {
      id: Date.now(),
      title: '新对话',
      messages: [{
        id: 'init',
        role: 'assistant',
        content: '抛光金属组件已就绪。光泽将随系统动态渲染。',
        timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
      }]
    };
    setSessions(prev => [newSession, ...prev]);
    setActiveSessionId(newSession.id);
    setShowHistoryDropdown(false);
  };

  const currentSession = sessions.find(s => s.id === activeSessionId) || { messages: [], title: '新对话' };

  const handleSend = () => {
    if (!input.trim() || !activeSessionId) return;
    
    const userMessage = {
      id: Date.now(),
      role: 'user',
      content: input,
      timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    };

    setSessions(prevSessions => {
      return prevSessions.map(session => {
        if (session.id === activeSessionId) {
          const isFirstUserMsg = session.messages.filter(m => m.role === 'user').length === 0;
          const updatedMessages = [...session.messages, userMessage];
          const updatedTitle = isFirstUserMsg 
            ? (input.length > 20 ? input.substring(0, 20) + '...' : input)
            : session.title;
          return { ...session, messages: updatedMessages, title: updatedTitle };
        }
        return session;
      });
    });

    setInput('');
    setIsTyping(true);

    setTimeout(() => {
      setSessions(prevSessions => {
        return prevSessions.map(session => {
          if (session.id === activeSessionId) {
            return { 
              ...session, 
              messages: [...session.messages, {
                id: Date.now() + 1,
                role: 'assistant',
                content: `[分析] 金属表面反射率正常。已进入[${activePersona}]模式。`,
                timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
              }] 
            };
          }
          return session;
        });
      });
      setIsTyping(false);
    }, 1200);
  };

  const deleteSession = (e, id) => {
    e.stopPropagation();
    const updated = sessions.filter(s => s.id !== id);
    setSessions(updated);
    if (activeSessionId === id) {
      if (updated.length > 0) setActiveSessionId(updated[0].id);
      else createNewSession();
    }
  };

  const onInputActive = () => {
    if (showHistoryDropdown) setShowHistoryDropdown(false);
  };

  return (
    <div className="flex h-screen w-full bg-[#f2f2f7] font-sans overflow-hidden relative text-neutral-800">
      <aside className="fixed top-0 left-0 h-full bg-neutral-50 flex flex-col w-[380px] z-50 border-r border-neutral-200 shadow-[2px_0_12px_rgba(0,0,0,0.03)]">
        
        {/* 顶部标题区 */}
        <div className="relative">
          <div className="h-14 flex items-center justify-between px-6 border-b border-neutral-200 bg-white/80 backdrop-blur-sm">
            <button 
              onClick={() => setShowHistoryDropdown(!showHistoryDropdown)}
              className="flex items-center space-x-2 group max-w-[85%] -ml-1 px-2 py-1.5 rounded-lg hover:bg-neutral-100 transition-colors"
            >
              <span className="text-sm font-semibold text-neutral-900 truncate">
                {currentSession.title}
              </span>
              <ChevronDown size={14} className={`text-neutral-400 transition-transform ${showHistoryDropdown ? 'rotate-180' : ''}`} />
            </button>
            <button onClick={createNewSession} className="text-neutral-400 hover:text-neutral-900 p-2 hover:bg-neutral-100 rounded-full">
              <Plus size={20} />
            </button>
          </div>

          {showHistoryDropdown && (
            <>
              <div className="fixed inset-0 z-[-1]" onClick={() => setShowHistoryDropdown(false)}></div>
              {/* 下拉面板：优化滚动视觉提示 */}
              <div className="absolute top-[56px] left-0 right-0 bg-white border-b border-neutral-200 shadow-2xl z-50 animate-in fade-in slide-in-from-top-2 duration-200">
                <div className="p-2 space-y-1 max-h-[360px] overflow-y-auto custom-scrollbar relative">
                  <div className="px-4 py-3 text-[10px] uppercase tracking-[0.2em] text-neutral-400 font-bold sticky top-0 bg-white/95 backdrop-blur-md z-10 border-b border-neutral-50 mb-1">
                    对话历史 ({sessions.length})
                  </div>
                  <div className="space-y-1">
                    {sessions.map(s => (
                      <button 
                        key={s.id} 
                        onClick={() => { setActiveSessionId(s.id); setShowHistoryDropdown(false); }}
                        className={`w-full flex items-center justify-between p-3 rounded-xl transition-all group ${activeSessionId === s.id ? 'bg-neutral-100' : 'hover:bg-neutral-50'}`}
                      >
                        <div className="flex items-center overflow-hidden flex-1">
                          <MessageSquare size={14} className={`mr-3 flex-shrink-0 ${activeSessionId === s.id ? 'text-blue-600' : 'text-neutral-300'}`} />
                          <span className={`text-xs truncate ${activeSessionId === s.id ? 'font-semibold text-neutral-900' : 'text-neutral-600'}`}>
                            {s.title}
                          </span>
                        </div>
                        <Trash2 
                          size={12} 
                          className="text-neutral-300 hover:text-red-500 opacity-0 group-hover:opacity-100 ml-2 transition-opacity" 
                          onClick={(e) => deleteSession(e, s.id)} 
                        />
                      </button>
                    ))}
                  </div>
                </div>
              </div>
            </>
          )}
        </div>

        <div className="flex-1 overflow-y-auto p-6 space-y-8 scrollbar-none bg-white">
          {currentSession.messages.map((msg) => (
            <div key={msg.id} className={`flex flex-col ${msg.role === 'user' ? 'items-end' : 'items-start'}`}>
              <div className={`max-w-[90%] text-[13px] leading-relaxed ${msg.role === 'user' ? 'text-blue-600 font-medium' : 'text-neutral-700'}`}>
                {msg.content}
              </div>
            </div>
          ))}
          {isTyping && <div className="text-[10px] text-neutral-400 animate-pulse font-mono tracking-widest uppercase">Assistant is analyzing...</div>}
          <div ref={messagesEndRef} />
        </div>

        <div className="p-6 pt-2 bg-white border-t border-neutral-100">
          <div className="input-case-polished" onPointerDown={onInputActive}>
            <div className="input-inner-container-polished">
              <textarea 
                ref={textareaRef}
                value={input}
                onChange={(e) => setInput(e.target.value)}
                onFocus={onInputActive}
                onKeyDown={(e) => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend(); } }}
                className="metal-textarea" 
                placeholder="探讨集体无意识..." 
                rows="1"
              />
              
              <div className="action-bar">
                <div className="dropdown-container">
                  {showPersonaDropdown && (
                    <div className="dropdown-menu active">
                      {personas.map(p => (
                        <div 
                          key={p} 
                          className="dropdown-item" 
                          onClick={() => { setActivePersona(p.split(' ')[0]); setShowPersonaDropdown(false); }}
                        >
                          {p}
                        </div>
                      ))}
                    </div>
                  )}
                  
                  <button 
                    className="persona-trigger"
                    onClick={(e) => { e.stopPropagation(); setShowPersonaDropdown(!showPersonaDropdown); }}
                  >
                    <span className="persona-title">{activePersona}</span>
                    <svg className={`chevron-icon transition-transform ${showPersonaDropdown ? 'rotate-180' : ''}`} viewBox="0 0 24 24">
                      <path d="M7.41 8.59L12 13.17l4.59-4.58L18 10l-6 6-6-6 1.41-1.41z"></path>
                    </svg>
                  </button>
                </div>

                <button 
                  className="mui-send-btn" 
                  onClick={handleSend}
                  disabled={!input.trim()}
                >
                  <svg className={`mui-send-icon ${input.trim() ? 'fill-[#1c1c1e]' : 'fill-[#8e8e93]'}`} viewBox="0 0 24 24">
                    <path d="M2.01 21L23 12 2.01 3 2 10l15 2-15 2z"></path>
                  </svg>
                </button>
              </div>
            </div>
          </div>
        </div>
      </aside>

      <main className="ml-[380px] flex-1 bg-[#f2f2f7] flex items-center justify-center relative">
        <div className="opacity-[0.05] select-none font-black text-8xl tracking-tighter text-black uppercase italic">
          Workspace
        </div>
      </main>

      <style dangerouslySetInnerHTML={{ __html: `
        :root {
          --polished-steel: linear-gradient(135deg, #8e8e93 0%, #ffffff 10%, #444446 25%, #f2f2f7 40%, #ffffff 50%, #3a3a3c 65%, #ffffff 85%, #8e8e93 100%);
        }

        @keyframes reflectRotate {
          0% { background-position: 0% 0%; }
          50% { background-position: 100% 100%; }
          100% { background-position: 0% 0%; }
        }

        .input-case-polished {
          position: relative;
          padding: 2.5px;
          border-radius: 20px; 
          background: var(--polished-steel);
          background-size: 200% 200%;
          box-shadow: 0 10px 30px -5px rgba(0, 0, 0, 0.15), 0 1px 1px rgba(255, 255, 255, 0.8) inset, 0 -1px 1px rgba(0, 0, 0, 0.2) inset;
          animation: reflectRotate 8s cubic-bezier(0.4, 0, 0.2, 1) infinite;
          transform: translateZ(0);
          will-change: background-position;
          width: 100%;
        }

        .input-inner-container-polished {
          background: #ffffff;
          border-radius: 17.5px;
          display: flex;
          flex-direction: column;
          position: relative;
          box-shadow: inset 0 2px 5px rgba(0, 0, 0, 0.08), inset 0 0 1px rgba(0, 0, 0, 0.1);
          contain: layout;
        }

        .metal-textarea {
          width: 100%;
          min-height: 44px;
          max-height: 300px;
          border: none;
          padding: 14px 16px 48px 16px; 
          font-size: 14px;
          line-height: 1.5;
          color: #1c1c1e;
          outline: none;
          resize: none;
          background: transparent;
          caret-color: #007aff;
        }

        .action-bar {
          position: absolute;
          bottom: 8px;
          left: 8px;
          right: 8px;
          height: 32px;
          display: flex;
          justify-content: space-between;
          align-items: center;
          z-index: 10;
        }

        .persona-trigger {
          display: flex;
          align-items: center;
          gap: 4px;
          background: #f2f2f7;
          height: 32px;
          padding: 0 10px;
          border-radius: 9px;
          cursor: pointer;
          transition: all 0.2s;
          border: 1px solid rgba(0,0,0,0.03);
          user-select: none;
        }
        .persona-trigger:hover { background: #e5e5ea; border-color: rgba(0,0,0,0.1); }
        .persona-title { font-size: 11px; font-weight: 600; color: #3a3a3c; }

        .dropdown-menu {
          position: absolute;
          bottom: calc(100% + 8px);
          left: 0;
          background: rgba(255, 255, 255, 0.95);
          backdrop-filter: blur(10px);
          border-radius: 12px;
          box-shadow: 0 10px 30px rgba(0, 0, 0, 0.15);
          min-width: 160px;
          padding: 4px;
          display: none;
          flex-direction: column;
          z-index: 50;
          border: 1px solid rgba(0,0,0,0.05);
        }
        .dropdown-menu.active { display: flex; }

        .dropdown-item {
          padding: 8px 12px;
          font-size: 12px;
          color: #3a3a3c;
          border-radius: 8px;
          cursor: pointer;
        }
        .dropdown-item:hover { background: rgba(0, 74, 255, 0.05); color: #007aff; }

        .mui-send-btn {
          width: 32px; height: 32px;
          border-radius: 50%;
          display: flex; align-items: center; justify-content: center;
          transition: background 0.2s, transform 0.1s;
        }
        .mui-send-btn:hover { background: rgba(0,0,0,0.05); }
        .mui-send-btn:active { transform: scale(0.92); }

        .mui-send-icon { width: 18px; height: 18px; transition: fill 0.2s; }
        .chevron-icon { width: 12px; height: 12px; fill: #8e8e93; }

        .custom-scrollbar::-webkit-scrollbar { width: 5px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background: #d1d1d6; border-radius: 10px; border: 1px solid white; }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover { background: #a2a2a7; }
      `}} />
    </div>
  );
};

export default App;
